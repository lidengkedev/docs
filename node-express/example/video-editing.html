app.js

import { createApp } from 'vue'
import ElementPlus from 'element-plus'
import zhCn from 'element-plus/locale/zh-cn'

const app = createApp({
  template: `
    <div class="video-editor">
      <h2>🎬 视频编辑工具</h2>

      <div class="btn-group">
        <el-button type="primary">上传视频</el-button>
        <el-button type="success">剪辑视频</el-button>
        <el-button type="warning">预览视频</el-button>
        <el-button type="danger">删除片段</el-button>
        <el-button>导出视频</el-button>
      </div>

      <el-input 
        placeholder="请输入视频名称" 
        class="video-input"
        clearable
      ></el-input>

      <el-divider>视频预览区域</el-divider>

      <el-card class="preview-card">
        <div>上传视频后在此处预览</div>
      </el-card>
    </div>
  `
})

app.use(ElementPlus, { locale: zhCn })
app.mount('#app')
